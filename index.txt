var log = $('#chat-log'), input = $('#chat-input'), sendBtn = $('#send-btn');
function addMsg(text, who, meta){
var d = document.createElement('div');
d.className = 'msg ' + who;
d.innerHTML = esc(text) + (meta ? '<div class="meta">' + esc(meta) + '</div>' : '');
log.appendChild(d);
log.scrollTop = log.scrollHeight;
return d;
}
async function send(){
var prompt = input.value.trim();
if (!prompt || sendBtn.disabled) return;
input.value = '';
addMsg(prompt, 'user');
sendBtn.disabled = true;
var thinking = addMsg('正在思考…', 'ai');
try {
var r = await fetch('/api/chat', { method: 'POST', headers: {'Content-Type':'application/json'}, body: JSON.stringify({prompt: prompt}) });
var j = await r.json();
thinking.remove();
if (j.ok) {
var meta = '模型：' + j.model + ' · 来源：' + (j.source || '');
if (j.skill && j.skill !== 'chat') meta += ' · 技能：' + (SKILL_LABEL[j.skill] || j.skill);
if (j.saved_scripts && j.saved_scripts.length) meta += ' · 代码已存：' + j.saved_scripts.join('、');
addMsg(j.reply, 'ai', meta);
} else {
addMsg('出错了：' + (j.error || '未知错误'), 'ai');
}
} catch (e) {
thinking.remove();
addMsg('连不上本地服务，请确认 App 还在运行。', 'ai');
}
sendBtn.disabled = false;
input.focus();
}
sendBtn.addEventListener('click', send);
var composing = false;
input.addEventListener('compositionstart', function(){ composing = true; });
input.addEventListener('compositionend', function(){ composing = false; });
input.addEventListener('keydown', function(e){
  if (composing || e.isComposing || e.keyCode === 229) return;
  if (e.key === 'Enter' && e.shiftKey) return;
  if (e.key === 'Enter') send();
});
async function loadMemories(q){
var mode = ($('#search-mode') && $('#search-mode').value) || 'keyword';
var url = q ? '/api/search?q=' + encodeURIComponent(q) + '&mode=' + mode : '/api/memories';
var r = await fetch(url); var j = await r.json();
var hint = $('#semantic-hint');
if (hint) {
if (!q) { hint.textContent = ''; }
else if (j.mode === 'semantic') { hint.textContent = '按意思找的结果（语义搜索）。'; }
else if (j.mode === 'keyword_fallback') { hint.textContent = '语义搜索不可用（没装 BGE-M3 嵌入模型），已退回关键词搜索。'; }
else if (j.mode === 'hybrid') { hint.textContent = '混合搜索：语义 + 关键词。'; }
else { hint.textContent = ''; }
}
var list = $('#memory-list');
if (!j.ok || !j.memories.length) { list.innerHTML = '<div class="empty">暂无记忆</div>'; return; }
list.innerHTML = j.memories.map(function(m){
var t = KIND_TAG[m.kind] || [m.kind, ''];
var score = (typeof m.score === 'number') ? ' <span class="time">相似度 ' + m.score.toFixed(2) + '</span>' : '';
return '<div class="card"><span class="tag ' + t[1] + '">' + esc(t[0]) + '</span>'
+ '<span class="time">' + esc(m.created_at) + '</span>' + score + '<div>' + esc(m.content) + '</div></div>';
}).join('');
}

$('#memory-refresh').addEventListener('click', function(){ $('#search-input').value = ''; loadMemories(); });

var currentPlanId = null;
function renderPlan(plan){
var el = $('#plan-current');
if (!plan) { el.innerHTML = ''; return; }
var done = plan.steps.filter(function(s){ return s.status === 'done'; }).length;
var html = '<div class="card"><div style="font-weight:600;">目标：' + esc(plan.goal) + '</div>'
+ '<div class="time">创建：' + esc(plan.created_at) + ' · 状态：' + (plan.status === 'done' ? '已完成' : plan.status === 'running' ? '进行中' : '草稿') + ' · ' + done + '/' + plan.steps.length + ' 步</div>';
plan.steps.forEach(function(s){
var tag = s.status === 'done' ? '<span class="tag done">已完成</span>' : '<span class="tag wait">待执行</span>';
html += '<div style="margin-top:10px;">' + tag + '<b>步骤 ' + (s.idx + 1) + '：</b>' + esc(s.text) + '</div>';
if (s.result) html += '<pre class="result">' + esc(s.result) + '</pre>';
});
html += '<div class="row" style="margin-top:12px;">'
+ '<button class="btn" id="plan-run-step"' + (done >= plan.steps.length ? ' disabled' : '') + '>执行下一步</button>'
+ '<button class="btn secondary" id="plan-reflect"' + (done < plan.steps.length ? ' disabled' : '') + '>自我检查</button>'
+ '</div><div id="plan-reflection"></div></div>';
el.innerHTML = html;
var runBtn = $('#plan-run-step');
if (runBtn) runBtn.addEventListener('click', async function(){
runBtn.disabled = true;
runBtn.textContent = '执行中…';
try {
var r = await fetch('/api/plan/step', { method: 'POST', headers: {'Content-Type':'application/json'}, body: JSON.stringify({plan_id: plan.id}) });
var j = await r.json();
if (j.ok) {
renderPlan(j.plan);
if (j.step && j.step.proposals && j.step.proposals.length) alert('这一步想写 ' + j.step.proposals.length + ' 个文件，已放进“待审批”，请检查后批准。');
loadPending();
} else {

async function loadPending(){
var r = await fetch('/api/pending'); var j = await r.json();
var list = $('#pending-list');
if (!j.ok || !j.pending.length) { list.innerHTML = '<div class="empty">没有待审批的文件</div>'; return; }
list.innerHTML = j.pending.map(function(p){
var src = (p.plan_id === 0 && p.step_idx === -1) ? '文档导入' : ('任务#' + p.plan_id + ' 第 ' + (p.step_idx + 1) + ' 步');
return '<div class="card"><b>' + esc(p.filename) + '</b>'
+ '<span class="time"> · ' + src + ' · ' + esc(p.created_at) + '</span>'
+ '<pre class="result">' + esc(p.content.slice(0, 500)) + (p.content.length > 500 ? '\n……（共 ' + p.content.length + ' 字）' : '') + '</pre>'
+ '<div class="row" style="margin-top:8px;margin-bottom:0;">'
+ '<button class="btn" data-act="approve" data-id="' + p.id + '">批准写入</button>'
+ '<button class="btn warn" data-act="reject" data-id="' + p.id + '">拒绝</button>'
+ '</div></div>';
}).join('');
list.querySelectorAll('button').forEach(function(b){
b.addEventListener('click', async function(){
var act = b.dataset.act, id = parseInt(b.dataset.id, 10);
var r = await fetch('/api/pending/' + act, { method: 'POST', headers: {'Content-Type':'application/json'}, body: JSON.stringify({id: id}) });
var j = await r.json();
alert(j.message || (j.ok ? '已处理' : '处理失败'));
loadPending();
loadFiles();
refreshStatus();
});
});
}
async function loadInbox(){
var r = await fetch('/api/inbox'); var j = await r.json();
var list = $('#inbox-list');
if (!j.ok || !j.files.length) { list.innerHTML = '<div class="empty">收件箱是空的。把文档放进 App 目录下的 inbox/ 文件夹，再点"刷新收件箱"。</div>'; return; }
list.innerHTML = j.files.map(function(f){

+ '<span class="time"> · ' + (f.size/1024).toFixed(1) + ' KB</span>'
+ ' <button class="btn" data-ingest="' + esc(f.name) + '">导入（转 Markdown）</button></div>';
}).join('');

b.disabled = false;
if (j.ok) { loadInbox(); loadPending(); }
});
});
}
